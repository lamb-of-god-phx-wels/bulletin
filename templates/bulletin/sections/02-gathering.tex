\bulletinheading{Confession of Sins}

\begin{liturgy}
\minister{Let us confess our sins to the Lord our God.}
\congregation{Lord God, we have sinned against you in thought, word, and deed. We have not loved you with our whole heart; we have not loved our neighbors as ourselves. We are truly sorry and repent of all our sins. For the sake of your Son, Jesus Christ, have mercy on us and forgive us, that we may delight in your will and walk in your ways, to the glory of your name. Amen.}
\minister{Almighty God, in his mercy, has given his Son to die for you and for his sake forgives you all your sins. As a called servant of Christ, I therefore forgive you all your sins in the name of the Father and of the Son and of the Holy Spirit.}
\congregation{Amen.}
\end{liturgy}
