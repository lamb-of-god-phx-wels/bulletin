\bulletinheading{Prayer of the Church}

\begin{liturgy}
\minister{Lord Jesus Christ, Son of God and Mary's son, in the fullness of time you came into our world to save us from sin and death:}
\prayerresponse{You ushered in the day of grace so long foretold.}

\minister{Beloved Son of the Father, revered by the Magi, baptized by John, you came preaching and teaching, healing and comforting, forgiving and encouraging:}
\prayerresponse{You brought the light of life to those walking in darkness, and the joy of salvation to those doomed to death.}

\minister{Prince of Peace, shine like the sunrise for us and the people of our world. Let the good news of salvation be heard in the remotest corners of the earth.}
\prayerresponse{Arouse us and our missionaries to flood the world with the light of your gospel.}

\rubric{Special prayers and petitions may be offered.}

\minister{Hear us, Lord, as we bring you our private petitions.}

\silentprayer
\end{liturgy}

\bulletinheading{The Lord's Prayer}

\begin{liturgy}
\congregation{Our Father in heaven, hallowed be your name, your kingdom come,
your will be done on earth as in heaven. Give us today our daily bread. Forgive us our sins,
as we forgive those who sin against us. Lead us not into temptation, but deliver us from evil.
For the kingdom, the power, and the glory are yours now and forever. Amen.}
\end{liturgy}

\bulletinheading{The Lord's Blessing}

\begin{liturgy}
\minister{Brothers and sisters, go in peace. Live in harmony with one another. Serve the Lord with gladness.}
\minister{The Lord bless you and keep you. The Lord make his face shine on you and be gracious to you.
The Lord look on you with favor and give you peace.}
\congregation{Amen.}
\end{liturgy}

\hymn{Closing Hymn}{As With Gladness Men of Old (CW 83)}

\begin{hymnverses}
\verseitem{As with gladness men of old did the guiding star behold,
as with joy they hailed its light, leading onward, beaming bright,
so, most gracious Lord, may we evermore your followers be.}

\verseitem{As with joyful steps they sped, Savior, to your lowly bed,
there to bend the knee before you whom heaven and earth adore,
so may we with willing feet ever seek your mercy seat.}

\verseitem{Holy Jesus, every day keep us in the narrow way;
and, when earthly things are past, bring our ransomed souls at last
where they need no star to guide, where no clouds your glory hide.}
\end{hymnverses}
